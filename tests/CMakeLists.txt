function(ae_suite name)
    cmake_parse_arguments(SUITE "CHATLINK" "" "CASES" ${ARGN})
    set(declarations "")
    set(dispatch "")
    set(sources "")
    foreach(case IN LISTS SUITE_CASES)
        string(APPEND declarations "int case_${case}(int, char**);\n")
        string(APPEND dispatch "    if (name == \"${case}\") return case_${case}(argc - 1, argv + 1);\n")
        list(APPEND sources "${case}.cpp")
    endforeach()
    configure_file(suite_main.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/${name}_main.cpp" @ONLY)
    add_executable(ae_${name} "${CMAKE_CURRENT_BINARY_DIR}/${name}_main.cpp" ${sources})
    target_include_directories(ae_${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../lib")
    # Assertions are part of the test behavior, including in Release builds.
    if(MSVC)
        target_compile_options(ae_${name} PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(ae_${name} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    endif()
    if(SUITE_CHATLINK)
        target_sources(ae_${name} PRIVATE ../src/ChatLink.cpp)
    endif()
    # Run each case in a fresh process; preserve isolation and CTest diagnostics.
    foreach(case IN LISTS SUITE_CASES)
        add_test(NAME "${name}.${case}" COMMAND ae_${name} "${case}")
        set_tests_properties("${name}.${case}" PROPERTIES LABELS "${name}")
    endforeach()
endfunction()
